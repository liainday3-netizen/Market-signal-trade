    return ohlcv
🧮 STEP 5 — FEATURE ENGINEERING

features.py

import pandas as pd

def create_features(data):
    df = pd.DataFrame(data, columns=["time","open","high","low","close","volume"])
    
    df["returns"] = df["close"].pct_change()
    df["ma"] = df["close"].rolling(10).mean()
    df["volatility"] = df["returns"].rolling(10).std()
    
    df = df.dropna()
    return df
🤖 STEP 6 — AI MODEL (SIMPLE START)

model.py

import torch
import torch.nn as nn

class Model(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc = nn.Sequential(
            nn.Linear(3, 32),
            nn.ReLU(),
            nn.Linear(32, 1),
            nn.Sigmoid()
        )

    def forward(self, x):
        return self.fc(x)

model = Model()
📊 STEP 7 — STRATEGY

strategy.py

import torch

def generate_signal(model, df):
    latest = df.iloc[-1][["returns","ma","volatility"]].values
    x = torch.tensor(latest, dtype=torch.float32)
    
    pred = model(x)
    
    if pred > 0.6:
        return "BUY"
    elif pred < 0.4:
        return "SELL"
    else:
        return "HOLD"

main.py

from data import get_data
from features import create_features
from model import model
from strategy import generate_signal
from execution import execute_trade
from config import SYMBOL, TIMEFRAME

while True:
    data = get_data(SYMBOL, TIMEFRAME)
    df = create_features(data)
    
    signal = generate_signal(model, df)
    
    print("Signal:", signal)
    
    execute_trade(signal, SYMBOL)
